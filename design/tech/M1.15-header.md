```ts
/**
 * The one route registry for the header and the footer (footer.md §3.2,
 * header.md §3.2). A link is rendered only when its route is built: flip `live`
 * in the task that builds the route, and the link appears in both places with no
 * other change. Row order is the display order in every placement.
 *
 * Plain data and pure functions with no imports, so Server Components, client
 * islands and Node (Playwright specs) can all import it.
 */

export type FooterGroup = "explore" | "help" | "legal";
export type HeaderPlacement = "primary" | "more" | "account" | "bag";

export type RouteEntry = {
  key: string;
  /** Visible text (footer, drawer); the header's icon buttons use it as their accessible name. */
  label: string;
  href: string;
  /** The task that builds the route; null = no task yet. */
  milestone: string | null;
  /** True once the route is built. */
  live: boolean;
  footer?: FooterGroup;
  header?: HeaderPlacement;
};

// One row per route, as in the footer.md §3.2.1 table.
// prettier-ignore
export const SITE_ROUTES: readonly RouteEntry[] = [
  { key: "shop", label: "Shop", href: "/shop", milestone: "M3.5", live: false, footer: "explore", header: "primary" },
  { key: "collections", label: "Collections", href: "/collections", milestone: "M3.7", live: false, footer: "explore", header: "primary" },
  { key: "gift-sets", label: "Gift sets", href: "/collections/gift-sets", milestone: "M3.7", live: false, footer: "explore", header: "primary" },
  { key: "our-story", label: "Our story", href: "/our-story", milestone: "M8.2", live: false, footer: "explore", header: "primary" },
  { key: "search", label: "Search", href: "/search", milestone: null, live: false, footer: "explore", header: "more" },
  { key: "faq", label: "FAQ", href: "/faq", milestone: "M8.2", live: false, footer: "help", header: "more" },
  { key: "contact", label: "Contact us", href: "/contact", milestone: "M8.2", live: false, footer: "help", header: "more" },
  { key: "shipping-returns", label: "Shipping & returns", href: "/shipping-and-returns", milestone: "M6.0b", live: false, footer: "help" },
  { key: "account", label: "Your account", href: "/account", milestone: "M5.7", live: false, footer: "help", header: "account" },
  { key: "sign-in", label: "Sign in", href: "/sign-in", milestone: "M5.2", live: false, footer: "help", header: "account" },
  { key: "bag", label: "Bag", href: "/bag", milestone: "M5.4", live: false, header: "bag" },
  { key: "privacy", label: "Privacy policy", href: "/privacy", milestone: "M2.5b", live: false, footer: "legal" },
  { key: "terms", label: "Terms & conditions", href: "/terms", milestone: "M6.0b", live: false, footer: "legal" },
];

/** Feature flags that are not routes (footer.md §3.2.2). */
export const features = {
  "collection-pages": { milestone: "M3.7", live: false }, // "Our collections" from getCollections()
  newsletter: { milestone: "M2.7", live: false }, // the footer teaser slot
} as const;

export type RegistryOptions = {
  /** Dev preview pages only: treat every entry as live. */
  preview?: boolean;
};

const isLive = (entry: RouteEntry, options: RegistryOptions): boolean =>
  options.preview === true || entry.live;

const byKey = (key: string): RouteEntry | undefined =>
  SITE_ROUTES.find((entry) => entry.key === key);

const placed = (
  placement: HeaderPlacement,
  options: RegistryOptions,
): RouteEntry[] =>
  SITE_ROUTES.filter(
    (entry) => entry.header === placement && isLive(entry, options),
  );

/** Shop, Collections, Gift sets, Our story: whichever are live, in that order. */
export function headerPrimary(options: RegistryOptions = {}): RouteEntry[] {
  return placed("primary", options);
}

/** Search, FAQ, Contact us (the drawer's More nav). */
export function headerMore(options: RegistryOptions = {}): RouteEntry[] {
  return placed("more", options);
}

/** "Your account" when signed in, "Sign in" when not; null if that route isn't live. */
export function headerAccount(
  signedIn: boolean,
  options: RegistryOptions = {},
): RouteEntry | null {
  const entry = byKey(signedIn ? "account" : "sign-in");
  return entry && isLive(entry, options) ? entry : null;
}

export function headerBag(options: RegistryOptions = {}): RouteEntry | null {
  const entry = byKey("bag");
  return entry && isLive(entry, options) ? entry : null;
}

const ACCOUNT_SLOT = new Set(["account", "sign-in"]);

/**
 * The live entries of one footer group, in row order. Help ends with the account
 * slot: `account` if live, else `sign-in` if live, never both (footer.md §3.2.1).
 */
export function footerGroup(
  group: FooterGroup,
  options: RegistryOptions = {},
): RouteEntry[] {
  const entries = SITE_ROUTES.filter(
    (entry) =>
      entry.footer === group &&
      !ACCOUNT_SLOT.has(entry.key) &&
      isLive(entry, options),
  );
  if (group === "help") {
    const slot = [byKey("account"), byKey("sign-in")].find(
      (entry) => entry?.footer === "help" && isLive(entry, options),
    );
    if (slot) entries.push(slot);
  }
  return entries;
}

/**
 * The most specific href that matches the path: the path itself, or a parent
 * of it (`/collections/moonlit-terrace` matches `/collections`). Only one link
 * of a set is current at a time (header.md §7.1).
 */
export function currentHref(
  pathname: string | null,
  hrefs: readonly string[],
): string | null {
  if (!pathname) return null;
  let best: string | null = null;
  for (const href of hrefs) {
    const matches = pathname === href || pathname.startsWith(`${href}/`);
    if (matches && (best === null || href.length > best.length)) best = href;
  }
  return best;
}
```

```tsx
"use client";

import Link from "next/link";
import { usePathname } from "next/navigation";
import { useContext } from "react";

import { FourPointStarIcon } from "@/components/icons/glyphs";
import { currentHref } from "@/components/layout/site-routes";

import { HeaderPreviewContext } from "./preview-context";

/**
 * bar: the Primary nav in the bar (lg+). drawer: the drawer's Primary links.
 * more: the drawer's More links. M1.17 adds "footer", which reads
 * FooterPreviewContext instead.
 */
export type NavLinkVariant = "bar" | "drawer" | "more";

const PREVIEW_CONTEXT = {
  bar: HeaderPreviewContext,
  drawer: HeaderPreviewContext,
  more: HeaderPreviewContext,
} as const;

// header.md §4.2, §7.2: no underline at rest; hover 1px currentColor; current
// 2px gold-300. The hover rule only adds the line, so current + hover keeps 2px gold.
const BAR =
  "inline-flex min-h-(--size-touch) items-center px-3 type-nav whitespace-nowrap text-text-secondary decoration-current decoration-1 underline-offset-[0.4em] transition-colors hover:text-text-primary hover:underline aria-[current=page]:text-text-primary aria-[current=page]:underline aria-[current=page]:decoration-link aria-[current=page]:decoration-2";

// header.md §5.6, §7.3: the current link gains the star; forced colors add an underline.
const DRAWER =
  "flex min-h-14 w-full items-center gap-3 type-h3 text-text-primary transition-colors hover:text-link forced-colors:aria-[current=page]:underline";
const MORE =
  "flex min-h-(--size-touch) w-full items-center gap-3 type-nav text-text-secondary transition-colors hover:text-link aria-[current=page]:text-text-primary forced-colors:aria-[current=page]:underline";

const CLASS = { bar: BAR, drawer: DRAWER, more: MORE } as const;

type NavLinkProps = {
  href: string;
  label: string;
  variant: NavLinkVariant;
  /** Every href of the link set this one belongs to: only the most specific match is current. */
  peers: readonly string[];
  /** /dev/header: a plain anchor (nothing prefetched) and the preview's current path. */
  preview?: boolean;
};

export function NavLink({
  href,
  label,
  variant,
  peers,
  preview = false,
}: NavLinkProps) {
  const pathname = usePathname();
  const previewState = useContext(PREVIEW_CONTEXT[variant]);
  const path = preview ? (previewState?.currentPath ?? null) : pathname;
  const current = currentHref(path, peers) === href;
  const props = {
    href,
    className: CLASS[variant],
    "aria-current": current ? ("page" as const) : undefined,
  };
  const content =
    variant === "bar" ? (
      label
    ) : (
      <>
        {current && <FourPointStarIcon className="size-3 shrink-0 text-link" />}
        <span>{label}</span>
      </>
    );
  return preview ? (
    <a {...props}>{content}</a>
  ) : (
    <Link {...props}>{content}</Link>
  );
}
```

```tsx
/** False in the server render and the hydration pass, true after. */
function useHydrated(): boolean {
  return useSyncExternalStore(
    subscribe,
    () => true,
    () => false,
  );
}

function bagName(label: string, count: number): string {
  if (count === 0) return `${label}, empty`;
  return `${label}, ${NUMBER.format(count)} ${count === 1 ? "item" : "items"}`;
}

function announcement(count: number): string {
  if (count === 0) return "Your bag is empty";
  return `Your bag has ${NUMBER.format(count)} ${count === 1 ? "item" : "items"}`;
}

type BagButtonProps = {
  href: string;
  /** The registry label, "Bag": the name until the count is known. */
  label: string;
  preview?: boolean;
};

/**
 * The bag link, its gold count badge and a polite live region (header.md §6).
 * The server render and the hydration pass show "Bag" and no badge; the count is
 * read after hydration (M1.15: only from the /dev/header preview; M5.4 connects
 * the cart store). The first read is never announced; later changes are, once
 * they have settled for 500ms.
 */
export function BagButton({ href, label, preview = false }: BagButtonProps) {
  const previewState = useContext(HeaderPreviewContext);
  const hydrated = useHydrated();
  const count = hydrated && previewState ? previewState.bagCount : null;

  const [status, setStatus] = useState("");
  const lastRead = useRef<number | null>(null);
  useEffect(() => {
    if (count === null || count === lastRead.current) return;
    const first = lastRead.current === null;
    lastRead.current = count;
    if (first) return;
    const timer = setTimeout(
      () => setStatus(announcement(count)),
      ANNOUNCE_DELAY,
    );
    return () => clearTimeout(timer);
  }, [count]);

  const props = {
    href,
    className: ICON_BUTTON,
    "aria-label": count === null ? label : bagName(label, count),
  };
  const content = (
    <>
      <BagIcon className={ICON_GLYPH} />
      {count !== null && count > 0 && (
        <span
          aria-hidden="true"
          className="absolute end-0.5 top-1 inline-flex min-h-5 min-w-5 items-center justify-center rounded-pill border border-transparent bg-badge-count px-1.5 type-caption font-medium text-text-inverse tabular-nums"
        >
          {count > 99 ? "99+" : count}
        </span>
      )}
    </>
  );
  return (
    <>
      {preview ? (
        <a {...props}>{content}</a>
      ) : (
        <Link {...props}>{content}</Link>
      )}
      <span role="status" data-print="hide" className="sr-only">
        {status}
      </span>
    </>
  );
}
```

```tsx
export const MENU_ID = "site-menu";
const LG = "(min-width: 64rem)";

/** Invoker commands open and close the dialog natively, before hydration too. */
const nativeInvokers = (): boolean =>
  "commandForElement" in HTMLButtonElement.prototype;

const inside = (rect: DOMRect, event: MouseEvent): boolean =>
  event.clientX >= rect.left &&
  event.clientX <= rect.right &&
  event.clientY >= rect.top &&
  event.clientY <= rect.bottom;

/**
 * The menu button and the drawer, a top-layer <dialog> opened with showModal()
 * (header.md §5). The dialog is in the server HTML (closed), so aria-controls
 * always resolves; its links are server-rendered children. aria-expanded follows
 * the dialog's own toggle and close events, so every way of opening and closing
 * keeps it right. Every close moves focus explicitly: to the menu button, or to
 * the home link when the viewport has grown to lg and the button is hidden.
 */
export function MobileMenu({ children }: { children: React.ReactNode }) {
  const buttonRef = useRef<HTMLButtonElement>(null);
  const dialogRef = useRef<HTMLDialogElement>(null);
  const [expanded, setExpanded] = useState(false);
  const pathname = usePathname();

  useEffect(() => {
    const dialog = dialogRef.current;
    const button = buttonRef.current;
    if (!dialog || !button) return;
    const lg = matchMedia(LG);
    let downOutside = false;

    const onToggle = (event: Event) =>
      setExpanded((event as ToggleEvent).newState === "open");
    const onClose = () => {
      setExpanded(false);
      const home = button
        .closest("header")
        ?.querySelector<HTMLElement>("[data-home-link]");
      (lg.matches ? home : button)?.focus();
    };
    // The scrim: close only when the press starts and ends outside the panel.
    const onPointerDown = (event: PointerEvent) => {
      downOutside = !inside(dialog.getBoundingClientRect(), event);
    };
    const onClick = (event: MouseEvent) => {
      const outside =
        downOutside && !inside(dialog.getBoundingClientRect(), event);
      downOutside = false;
      const link = (event.target as Element).closest("a[href]");
      if (outside || link) dialog.close();
    };
    // At lg the drawer can't stay open: close at once, without the exit motion.
    const onBreakpoint = () => {
      if (!lg.matches || !dialog.open) return;
      dialog.setAttribute("data-instant", "");
      dialog.close();
      requestAnimationFrame(() => dialog.removeAttribute("data-instant"));
    };

    setExpanded(dialog.open);
    dialog.addEventListener("toggle", onToggle);
    dialog.addEventListener("close", onClose);
    dialog.addEventListener("pointerdown", onPointerDown);
    dialog.addEventListener("click", onClick);
    lg.addEventListener("change", onBreakpoint);
    return () => {
      dialog.removeEventListener("toggle", onToggle);
      dialog.removeEventListener("close", onClose);
      dialog.removeEventListener("pointerdown", onPointerDown);
      dialog.removeEventListener("click", onClick);
      lg.removeEventListener("change", onBreakpoint);
    };
  }, []);

  // Back/forward or any other route change closes the drawer.
  const shownPath = useRef(pathname);
  useEffect(() => {
    if (shownPath.current === pathname) return;
    shownPath.current = pathname;
    dialogRef.current?.close();
  }, [pathname]);

  const open = () => {
    const dialog = dialogRef.current;
    if (dialog && !dialog.open && !nativeInvokers()) dialog.showModal();
  };
  const close = () => {
    if (!nativeInvokers()) dialogRef.current?.close();
  };

  return (
    <>
      <button
        ref={buttonRef}
        type="button"
        aria-label="Menu"
        aria-expanded={expanded}
        aria-controls={MENU_ID}
        aria-haspopup="dialog"
        commandfor={MENU_ID}
        command="show-modal"
        data-print="hide"
        onClick={open}
        className={`${ICON_BUTTON} lg:hidden noscript:hidden`}
      >
        <MenuIcon className={ICON_GLYPH} />
      </button>
      <dialog
        ref={dialogRef}
        id={MENU_ID}
        aria-label="Menu"
        className="site-menu fixed inset-y-0 start-auto end-0 m-0 h-auto max-h-none w-[min(22rem,100%-3rem)] max-w-none overflow-y-auto overscroll-contain rounded-s-lg border-0 border-s border-transparent bg-bg-overlay ps-6 pe-(--page-inline-end) pt-[env(safe-area-inset-top,0px)] pb-[calc(--spacing(6)+env(safe-area-inset-bottom,0px))] text-text-primary shadow-overlay"
      >
        <div className="flex min-h-full flex-col">
          <div className="flex h-(--header-h) shrink-0 items-center justify-end">
            <button
              type="button"
              aria-label="Close menu"
              commandfor={MENU_ID}
              command="close"
              onClick={close}
              className={ICON_BUTTON}
            >
              <CloseIcon className={ICON_GLYPH} />
            </button>
          </div>
          {children}
        </div>
      </dialog>
    </>
  );
}
```

```tsx
"use client";

import { useEffect, useRef } from "react";

/**
 * The header's at-rest / scrolled state (layout-shell.md §6.2, header.md §8.1):
 * `data-scrolled="true"` on the header once the page is scrolled past the top
 * 8px, from an IntersectionObserver on an 8px sentinel (no scroll listener).
 * Rendered as the first child of <main>, which starts right below the header, and
 * pulled up by the header's height to the top of the document, so the body keeps
 * its landmark order. Out of flow, empty and hidden: it never affects layout.
 */
export function HeaderScrollState() {
  const sentinelRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    const sentinel = sentinelRef.current;
    const header = document.querySelector<HTMLElement>("body > header");
    if (!sentinel || !header) return;
    const set = (scrolled: boolean) => {
      if (scrolled) header.setAttribute("data-scrolled", "true");
      else header.removeAttribute("data-scrolled");
    };
    // One read on mount, so a restored scroll position is right at once.
    set(sentinel.getBoundingClientRect().bottom < 0);
    const observer = new IntersectionObserver(([entry]) => {
      if (entry) set(!entry.isIntersecting);
    });
    observer.observe(sentinel);
    return () => observer.disconnect();
  }, []);

  return (
    <div
      ref={sentinelRef}
      aria-hidden="true"
      data-scroll-sentinel=""
      className="pointer-events-none invisible absolute start-0 top-[calc(-1*(var(--header-h)+env(safe-area-inset-top,0px)))] h-2 w-px"
    />
  );
}
```

## 4. Notes and pushback

1. **PUSHBACK (minor, design-agent wording): where the sentinel lives.** layout-shell.md §6.2 says "an 8px sentinel at the top of the document" and header.md §13.1 lists `HeaderScrollState` as a header island. It is built as the first child of `<main>`, positioned up to the document top (§1), because no sentinel inside the sticky header can track the scroll position, and a new `<body>` child would break the §2.1 body order. Behaviour is exactly as specced. Proposed wording for §6.2 / §13.1: "rendered at the start of `main`, positioned at the top of the document".
2. **Finding (no change asked):** the threshold is "more than 8px" (at exactly 8px an edge-adjacent sentinel still intersects); 1px, invisible.
3. **Finding for M1.17:** `footerGroup()` and `currentHref()` are already in the registry; `NavLink` takes `peers` (the footer passes every footer href, including the "Our collections" links, so `/collections/moonlit-terrace` marks Moonlit Terrace, not Collections) and gets a `footer` entry in `PREVIEW_CONTEXT`.
